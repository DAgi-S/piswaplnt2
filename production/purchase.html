Current Database Schema
Existing Table Structures:

-- Existing suppliers table
CREATE TABLE suppliers (
    id INT PRIMARY KEY AUTO_INCREMENT,
    company_name VARCHAR(255),
    contact_person VARCHAR(255),
    phone VARCHAR(50),
    address TEXT,
    active INT,
    created_at TIMESTAMP,
    updated_at TIMESTAMP
);

-- Existing purchases table
CREATE TABLE purchases (
    id INT PRIMARY KEY AUTO_INCREMENT,
    purchase_number VARCHAR(50),
    supplier_id INT,
    purchase_date DATE,
    sub_total DECIMAL(25,2),
    vat DECIMAL(25,2),
    grand_total DECIMAL(25,2),
    note TEXT,
    payment_status VARCHAR(20),
    paid_amount DECIMAL(25,2),
    status INT,
    active INT,
    created_by INT,
    created_at TIMESTAMP,
    updated_at TIMESTAMP,
    withholding_tax_enabled TINYINT,
    withholding_tax_amount DECIMAL(25,2)
);

-- Existing purchase_items table
CREATE TABLE purchase_items (
    id INT PRIMARY KEY AUTO_INCREMENT,
    purchase_id INT,
    product_id INT,
    quantity INT,
    rate DECIMAL(25,2),
    total DECIMAL(25,2),
    created_at TIMESTAMP,
    updated_at TIMESTAMP
);
