Required Table Structures:

-- Required sales_orders table
CREATE TABLE sales_orders (
    id INT PRIMARY KEY AUTO_INCREMENT,
    order_number VARCHAR(50),
    client_id INT,
    order_date DATE,
    delivery_date DATE,
    sub_total DECIMAL(25,2),
    vat DECIMAL(25,2),
    grand_total DECIMAL(25,2),
    payment_status VARCHAR(20),
    paid_amount DECIMAL(25,2),
    status INT,
    active INT,
    created_by INT,
    created_at TIMESTAMP,
    updated_at TIMESTAMP,
    FOREIGN KEY (client_id) REFERENCES clients(id)
);

-- Required sales_items table
CREATE TABLE sales_items (
    id INT PRIMARY KEY AUTO_INCREMENT,
    sales_order_id INT,
    product_id INT,
    quantity INT,
    rate DECIMAL(25,2),
    total DECIMAL(25,2),
    created_at TIMESTAMP,
    updated_at TIMESTAMP,
    FOREIGN KEY (sales_order_id) REFERENCES sales_orders(id),
    FOREIGN KEY (product_id) REFERENCES production_products(id)
);
